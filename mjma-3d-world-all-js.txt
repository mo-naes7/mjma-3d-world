function getGeminiClient() {
  const apiKey = process.env.GEMINI_API_KEY;
  if (!apiKey) {
    throw new Error("GEMINI_API_KEY is not configured on the server.");
  }
  return new GoogleGenAI({
    apiKey,
    httpOptions: {
      headers: {
        "User-Agent": "aistudio-build"
      }
    }
  });
}
async function translateBatchWithRetry(ai, targetLanguage, items, attempt = 0) {
  try {
    const response = await ai.models.generateContent({
      model: "gemini-4-hydrogen",
      contents: JSON.stringify({
        targetLanguage,
        items
      }),
      config: {
        systemInstruction: 'You are a professional e-commerce localization engine for "MJMA 3D World" (a 3D-printed dental training models and precision tools store). Translate EVERY single item text completely and naturally into the requested targetLanguage, including full product descriptions and customer reviews. Rules: 1) Keep "MJMA 3D World", "@dr_mjma", emails, phone numbers, and URLs untouched. 2) Keep interpolation placeholders like {count}, {currency}, {storeName}, {product}, {amount}, {email} exact and untranslated inside curly braces. 3) Return every id value unchanged and include all input items in translations. 4) Set dir strictly to "rtl" for right-to-left languages (Arabic, Hebrew, Persian, Urdu, Sorani Kurdish, Pashto, etc.) and "ltr" for all others. 5) Set languageCode to the lowercase ISO 639-1 / BCP-47 tag (e.g. "en", "tr", "fr", "de", "es", "ja", "zh", "ru").',
        thinkingConfig: { thinkingLevel: ThinkingLevel.LOW },
        responseMimeType: "application/json",
        responseSchema: {
          type: Type.OBJECT,
          properties: {
            languageCode: {
              type: Type.STRING,
              description: "BCP-47 language code for the target language."
            },
            dir: {
              type: Type.STRING,
              description: 'Text direction: "rtl" or "ltr".'
            },
            translations: {
              type: Type.ARRAY,
              items: {
                type: Type.OBJECT,
                properties: {
                  id: { type: Type.STRING },
                  text: { type: Type.STRING }
                },
                required: ["id", "text"]
              }
            }
          },
          required: ["languageCode", "dir", "translations"]
        }
      }
    });
    const raw = (response.text ?? "{}").trim();
    const parsed = JSON.parse(raw);
    const normalizedDir = parsed.dir === "rtl" ? "rtl" : "ltr";
    const normalizedCode = typeof parsed.languageCode === "string" && parsed.languageCode.trim() ? parsed.languageCode.trim().toLowerCase() : "en";
    const translations = Array.isArray(parsed.translations) ? parsed.translations : [];
    return {
      languageCode: normalizedCode,
      dir: normalizedDir,
      translations
    };
  } catch (err) {
    const status = typeof err === "object" && err !== null && "status" in err ? Number(err.status) : 0;
    const retriable = status === 429 || status >= 500 || status === 0;
    if (retriable && attempt < 2) {
      const waitMs = 400 * Math.pow(2, attempt);
      await new Promise((resolve) => setTimeout(resolve, waitMs));
      return translateBatchWithRetry(ai, targetLanguage, items, attempt + 1);
    }
    throw err;
  }
}
async function startServer() {
  const app = express();
  app.use(express.json({ limit: "1mb" }));
  app.post("/api/translate", async (req, res) => {
    try {
      const body = req.body;
      const targetLanguage = typeof body?.targetLanguage === "string" ? body.targetLanguage.trim().slice(0, 60) : "";
      if (!targetLanguage) {
        res.status(400).json({ error: "Target language is required." });
        return;
      }
      if (!Array.isArray(body?.items)) {
        res.status(400).json({ error: "Items array is required." });
        return;
      }
      const cleanItems = body.items.slice(0, 180).filter(
        (entry) => typeof entry === "object" && entry !== null && typeof entry.id === "string" && typeof entry.text === "string" && entry.text.trim().length > 0
      ).map((entry) => ({
        id: entry.id.slice(0, 120),
        text: entry.text.slice(0, 2500)
      }));
      if (cleanItems.length === 0) {
        res.json({ languageCode: "en", dir: "ltr", translations: {} });
        return;
      }
      const ai = getGeminiClient();
      const CHUNK_SIZE = 60;
      const chunks = [];
      for (let i = 0; i < cleanItems.length; i += CHUNK_SIZE) {
        chunks.push(cleanItems.slice(i, i + CHUNK_SIZE));
      }
      const batchResults = await Promise.all(
        chunks.map((chunk) => translateBatchWithRetry(ai, targetLanguage, chunk))
      );
      const mergedTranslations = {};
      for (const batch of batchResults) {
        for (const item of batch.translations) {
          if (item && typeof item.id === "string" && typeof item.text === "string") {
            mergedTranslations[item.id] = item.text;
          }
        }
      }
      res.json({
        languageCode: batchResults[0]?.languageCode || "en",
        dir: batchResults[0]?.dir || "ltr",
        translations: mergedTranslations
      });
    } catch (err) {
      const message = err instanceof Error ? err.message : "Translation request failed.";
      const status = typeof err === "object" && err !== null && "status" in err ? Number(err.status) || 500 : 500;
      res.status(status >= 400 && status < 600 ? status : 500).json({
        error: message
      });
    }
  });
  if (!IS_PRODUCTION) {
    const { createServer: createViteServer } = await import("vite");
    const vite = await createViteServer({
      server: { middlewareMode: true },
      appType: "spa"
    });
    app.use(vite.middlewares);
  } else {
    const distDir = path.resolve(__dirname, "dist");
    app.use(express.static(distDir));
    app.get("*", (_req, res) => {
      const indexHtml = path.join(distDir, "index.html");
      if (fs.existsSync(indexHtml)) {
        res.sendFile(indexHtml);
      } else {
        res.status(404).send("Build output not found. Run npm run build first.");
      }
    });
  }
  app.listen(PORT, "0.0.0.0", () => {
    console.log(`Server listening on http://0.0.0.0:${PORT}`);
  });
}
startServer().catch((err) => {
  console.error("Failed to start server:", err);
  process.exit(1);
});

import {
  createContext,
  useCallback,
  useContext,
  useEffect,
  useMemo,
  useRef,
  useState
} from "react";
import {
  AR_MESSAGES,
  PRESET_LANGUAGES,
  STATIC_BUNDLES
} from "./messages";
const LANG_PREF_KEY = "mjma_store_lang_v1";
const TRANSLATION_CACHE_KEY = "mjma_translations_cache_v3";
const DEFAULT_LANGUAGE = {
  target: "ar",
  label: "العربية",
  code: "ar",
  dir: "rtl"
};
const TranslationContext = createContext(null);
function normalizeKey(raw) {
  return raw.trim().toLowerCase();
}
function loadCache() {
  try {
    const raw = localStorage.getItem(TRANSLATION_CACHE_KEY);
    return raw ? JSON.parse(raw) : {};
  } catch {
    return {};
  }
}
function saveCache(cache) {
  try {
    localStorage.setItem(TRANSLATION_CACHE_KEY, JSON.stringify(cache));
  } catch {
  }
}
function loadSavedLang() {
  try {
    const raw = localStorage.getItem(LANG_PREF_KEY);
    if (!raw) return DEFAULT_LANGUAGE;
    const parsed = JSON.parse(raw);
    if (typeof parsed.target === "string" && parsed.target.trim()) {
      return {
        target: parsed.target,
        label: parsed.label || parsed.target,
        code: parsed.code || "ar",
        dir: parsed.dir === "ltr" ? "ltr" : "rtl"
      };
    }
    return DEFAULT_LANGUAGE;
  } catch {
    return DEFAULT_LANGUAGE;
  }
}
function interpolate(template, vars) {
  if (!vars) return template;
  return template.replace(/\{([a-zA-Z0-9_]+)\}/g, (match, name) => {
    return Object.prototype.hasOwnProperty.call(vars, name) ? String(vars[name]) : match;
  });
}
function formatArabicReviewCount(count) {
  if (count === 1) return AR_MESSAGES["reviews.oneReview"];
  if (count === 2) return "تقييمان";
  if (count <= 10) return `${count} تقييمات`;
  return `${count} تقييماً`;
}
function TranslationProvider({ children }) {
  const [language, setLanguage] = useState(() => loadSavedLang());
  const [cache, setCache] = useState(() => loadCache());
  const [isTranslating, setIsTranslating] = useState(false);
  const [translateError, setTranslateError] = useState("");
  const dynamicPoolRef = useRef(/* @__PURE__ */ new Set());
  const inFlightKeysRef = useRef(/* @__PURE__ */ new Set());
  const cacheRef = useRef(cache);
  cacheRef.current = cache;
  useEffect(() => {
    document.documentElement.lang = language.code;
    document.documentElement.dir = language.dir;
    try {
      localStorage.setItem(LANG_PREF_KEY, JSON.stringify(language));
    } catch {
    }
  }, [language]);
  const requestTranslationBatch = useCallback(
    async (targetLanguage, items) => {
      const response = await fetch("/api/translate", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ targetLanguage, items })
      });
      const data = await response.json();
      if (!response.ok) {
        throw new Error(data.error || "Translation request failed.");
      }
      return {
        languageCode: data.languageCode || "en",
        dir: data.dir === "rtl" ? "rtl" : "ltr",
        translations: data.translations || {}
      };
    },
    []
  );
  const translateMissingForLanguage = useCallback(
    async (target, displayLabel) => {
      const key = normalizeKey(target);
      if (key === "ar" || key === "arabic" || key === "العربية") {
        setTranslateError("");
        setLanguage(DEFAULT_LANGUAGE);
        return;
      }
      const preset = PRESET_LANGUAGES.find(
        (p) => normalizeKey(p.code) === key || normalizeKey(p.label) === key
      );
      const effectiveTarget = preset ? preset.label : target.trim();
      const effectiveKey = preset ? preset.code : key;
      const existingEntry = cacheRef.current[effectiveKey];
      const staticBundle = STATIC_BUNDLES[effectiveKey];
      const itemsToTranslate = [];
      const idMap = /* @__PURE__ */ new Map();
      let seq = 0;
      if (!staticBundle) {
        const cachedUi = existingEntry?.ui ?? {};
        for (const [msgKey, msgText] of Object.entries(AR_MESSAGES)) {
          if (!cachedUi[msgKey]) {
            const shortId = `u${seq++}`;
            idMap.set(shortId, { kind: "ui", key: msgKey });
            itemsToTranslate.push({ id: shortId, text: msgText });
          }
        }
      }
      const cachedDynamic = existingEntry?.dynamic ?? {};
      const inflightDynTokens = [];
      for (const dynText of dynamicPoolRef.current) {
        if (dynText.trim() && !cachedDynamic[dynText]) {
          const inflightToken = `${effectiveKey}::${dynText}`;
          if (!inFlightKeysRef.current.has(inflightToken)) {
            const shortId = `d${seq++}`;
            idMap.set(shortId, { kind: "dyn", sourceText: dynText });
            itemsToTranslate.push({ id: shortId, text: dynText });
            inFlightKeysRef.current.add(inflightToken);
            inflightDynTokens.push(inflightToken);
          }
        }
      }
      const immediateDir = preset?.dir ?? existingEntry?.dir ?? (staticBundle ? "ltr" : language.dir);
      const immediateCode = preset?.code ?? existingEntry?.code ?? effectiveKey;
      const immediateLabel = displayLabel || preset?.label || existingEntry?.label || effectiveTarget;
      setLanguage({
        target: effectiveKey,
        label: immediateLabel,
        code: immediateCode,
        dir: immediateDir
      });
      setTranslateError("");
      if (itemsToTranslate.length === 0) {
        return;
      }
      setIsTranslating(true);
      try {
        const result = await requestTranslationBatch(effectiveTarget, itemsToTranslate);
        setCache((prev) => {
          const prevLang = prev[effectiveKey] ?? {
            code: result.languageCode,
            dir: result.dir,
            label: immediateLabel,
            ui: {},
            dynamic: {}
          };
          const nextUi = { ...prevLang.ui };
          const nextDynamic = { ...prevLang.dynamic };
          for (const [itemId, translatedText] of Object.entries(result.translations)) {
            const mapping = idMap.get(itemId);
            if (!mapping || !translatedText?.trim()) continue;
            if (mapping.kind === "ui") {
              nextUi[mapping.key] = translatedText;
            } else {
              nextDynamic[mapping.sourceText] = translatedText;
            }
          }
          const updated = {
            ...prev,
            [effectiveKey]: {
              code: preset?.code ?? result.languageCode,
              dir: preset?.dir ?? result.dir,
              label: immediateLabel,
              ui: nextUi,
              dynamic: nextDynamic
            }
          };
          saveCache(updated);
          return updated;
        });
        setLanguage({
          target: effectiveKey,
          label: immediateLabel,
          code: preset?.code ?? result.languageCode,
          dir: preset?.dir ?? result.dir
        });
      } catch (err) {
        const msg = err instanceof Error ? err.message : "";
        setTranslateError(msg || AR_MESSAGES["lang.errorGeneric"]);
      } finally {
        for (const token of inflightDynTokens) {
          inFlightKeysRef.current.delete(token);
        }
        setIsTranslating(false);
      }
    },
    [language.dir, requestTranslationBatch]
  );
  const ensureDynamicTexts = useCallback(
    (texts) => {
      let hasNew = false;
      for (const raw of texts) {
        const clean = raw?.trim();
        if (clean && !dynamicPoolRef.current.has(clean)) {
          dynamicPoolRef.current.add(clean);
          hasNew = true;
        }
      }
      if (language.target === "ar") return;
      const currentCache = cacheRef.current[language.target]?.dynamic ?? {};
      const missing = [];
      for (const raw of texts) {
        const clean = raw?.trim();
        if (clean && !currentCache[clean] && !inFlightKeysRef.current.has(`${language.target}::${clean}`)) {
          missing.push(clean);
        }
      }
      if (missing.length > 0 || hasNew && !cacheRef.current[language.target]) {
        void translateMissingForLanguage(language.target, language.label);
      }
    },
    [language.label, language.target, translateMissingForLanguage]
  );
  const t = useCallback(
    (key, vars) => {
      if (language.target === "ar") {
        return interpolate(AR_MESSAGES[key], vars);
      }
      const staticBundle = STATIC_BUNDLES[language.target];
      if (staticBundle && staticBundle[key]) {
        return interpolate(staticBundle[key], vars);
      }
      const cached = cache[language.target]?.ui?.[key];
      if (cached) {
        return interpolate(cached, vars);
      }
      return interpolate(AR_MESSAGES[key], vars);
    },
    [cache, language.target]
  );
  const td = useCallback(
    (sourceText) => {
      const clean = sourceText?.trim();
      if (!clean || language.target === "ar") return sourceText;
      return cache[language.target]?.dynamic?.[clean] ?? sourceText;
    },
    [cache, language.target]
  );
  const formatReviewCount = useCallback(
    (count) => {
      if (language.target === "ar") {
        return formatArabicReviewCount(count);
      }
      if (count === 1) {
        return t("reviews.oneReview");
      }
      return t("reviews.manyReviews", { count });
    },
    [language.target, t]
  );
  const formatDate = useCallback(
    (ts) => {
      if (ts && typeof ts === "object" && "toDate" in ts && typeof ts.toDate === "function") {
        const locale = language.code === "ar" ? "ar-EG" : language.code;
        try {
          return ts.toDate().toLocaleDateString(locale, {
            year: "numeric",
            month: "short",
            day: "numeric"
          });
        } catch {
          return ts.toDate().toLocaleDateString("en-US", {
            year: "numeric",
            month: "short",
            day: "numeric"
          });
        }
      }
      return t("reviews.recently");
    },
    [language.code, t]
  );
  const clearTranslateError = useCallback(() => setTranslateError(""), []);
  const value = useMemo(
    () => ({
      language,
      isTranslating,
      translateError,
      setTargetLanguage: translateMissingForLanguage,
      clearTranslateError,
      t,
      td,
      formatReviewCount,
      formatDate,
      ensureDynamicTexts
    }),
    [
      clearTranslateError,
      ensureDynamicTexts,
      formatDate,
      formatReviewCount,
      isTranslating,
      language,
      t,
      td,
      translateError,
      translateMissingForLanguage
    ]
  );
  return <TranslationContext.Provider value={value}>
      {children}
    </TranslationContext.Provider>;
}
function useTranslation() {
  const ctx = useContext(TranslationContext);
  if (!ctx) {
    throw new Error("useTranslation must be used within TranslationProvider");
  }
  return ctx;
}
export {
  TranslationProvider,
  useTranslation
};

function generateSafeId(prefix) {
  const chars = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
  let rand = "";
  for (let i = 0; i < 8; i += 1) {
    rand += chars.charAt(Math.floor(Math.random() * chars.length));
  }
  return `${prefix}_${Date.now()}_${rand}`;
}
function generateOrderCode() {
  const chars = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
  let suffix = "";
  for (let i = 0; i < 4; i += 1) {
    suffix += chars.charAt(Math.floor(Math.random() * chars.length));
  }
  const num = Math.floor(1e3 + Math.random() * 9e3);
  return `ORD-${num}-${suffix}`;
}

  return {
    title: clampText(input.title, VALIDATION_LIMITS.PRODUCT_TITLE_MAX),
    category: clampText(input.category, VALIDATION_LIMITS.PRODUCT_CATEGORY_MAX),
    price: Math.min(1e6, Math.max(0.01, Number(input.price) || 1)),
    description: clampText(input.description, VALIDATION_LIMITS.PRODUCT_DESC_MAX),
    imageUrl: clampText(input.imageUrl, VALIDATION_LIMITS.PRODUCT_IMAGE_MAX),
    stock: Math.min(1e5, Math.max(0, Math.floor(Number(input.stock) || 0))),

  };
}

async function submitOrder(input) {
  const orderId = generateSafeId("ord");
  const orderNumber = generateOrderCode();
  const boundedItems = input.cartItems.slice(0, VALIDATION_LIMITS.ORDER_ITEMS_MAX).map((item) => ({
    productId: clampText(item.productId, 128),
    title: clampText(item.title, VALIDATION_LIMITS.PRODUCT_TITLE_MAX),
    variant: clampText(item.variant || "قياسي", 40),
    quantity: Math.min(100, Math.max(1, Math.floor(item.quantity))),
    price: Math.max(0.01, Number(item.price))
  }));
  const subtotal = Number(
    boundedItems.reduce((sum, item) => sum + item.price * item.quantity, 0).toFixed(2)
  );
  const shippingFee = Math.max(0, Number(input.shippingFee) || 0);
  const total = Number((subtotal + shippingFee).toFixed(2));
  const itemCount = boundedItems.reduce((sum, item) => sum + item.quantity, 0);
  const userId = auth.currentUser?.uid || "guest";
  try {
    const batch = writeBatch(db);
    batch.set(doc(db, "orders", orderId), {
      orderNumber,
      trackingCode: orderNumber,
      userId,
      customerName: clampText(input.customerName, VALIDATION_LIMITS.ORDER_NAME_MAX),
      customerPhone: clampText(input.customerPhone, VALIDATION_LIMITS.ORDER_PHONE_MAX),
      customerCity: clampText(input.customerCity, VALIDATION_LIMITS.ORDER_CITY_MAX),
      customerAddress: clampText(input.customerAddress, VALIDATION_LIMITS.ORDER_ADDRESS_MAX),
      notes: clampText(input.notes || "", VALIDATION_LIMITS.ORDER_NOTES_MAX),
      items: boundedItems,
      subtotal,
      shippingFee,
      total,
      paymentMethod: input.paymentMethod,
      status: "pending",
      createdAt: serverTimestamp(),
      updatedAt: serverTimestamp()
    });
    batch.set(doc(db, "order_tracking", orderNumber), {
      orderId,
      orderNumber,
      status: "pending",
      itemCount: Math.max(1, Math.floor(itemCount)),
      total,
      createdAt: serverTimestamp(),
      updatedAt: serverTimestamp()
    });
    await batch.commit();
    return { orderId, orderNumber };
  } catch (error) {
    handleFirestoreError(error, OperationType.WRITE, `orders/${orderId}`);
  }
}

import { useState } from "react";
import { Package } from "lucide-react";
import { resolveProductImageUrl } from "../data/seedCatalog";
function ResilientImage({
  src,
  alt,
  className = "",
  fallbackTitle
}) {
  const [failed, setFailed] = useState(false);
  const [isPortrait, setIsPortrait] = useState(false);
  const resolvedSrc = resolveProductImageUrl(src);
  if (!resolvedSrc || failed) {
    return <div
      className={`flex flex-col items-center justify-center bg-gradient-to-br from-[#2a1045] to-[#4c165c] text-pink-200/80 p-4 text-center ${className}`}
      role="img"
      aria-label={alt}
    >
        <Package className="w-8 h-8 mb-2 text-pink-300/75" aria-hidden="true" />
        {fallbackTitle && <span className="text-xs font-medium text-pink-100 line-clamp-2 max-w-[18ch]">
            {fallbackTitle}
          </span>}
      </div>;
  }
  const finalClassName = isPortrait && className.includes("object-cover") ? className.replace(
    "object-cover",
    "object-contain bg-gradient-to-b from-[#230b3d] via-[#17072b] to-[#120520]"
  ) : className;
  return <img
    src={resolvedSrc}
    alt={alt}
    referrerPolicy="no-referrer"
    onLoad={(e) => {
      const img = e.currentTarget;
      setIsPortrait(img.naturalHeight > img.naturalWidth);
    }}
    onError={() => setFailed(true)}
    className={finalClassName}
  />;
}

  ResilientImage
};

import { useEffect, useRef, useState } from "react";

function buildProductShareUrl(productId) {
  try {
    const url = new URL(window.location.href);
    url.searchParams.set("product", productId);
    url.hash = "";
    return url.toString();
  } catch {
    return `?product=${encodeURIComponent(productId)}`;
  }
}
async function copyTextToClipboard(text) {
  if (navigator.clipboard && typeof navigator.clipboard.writeText === "function") {
    try {
      await navigator.clipboard.writeText(text);
      return true;
    } catch {
    }
  }
  try {
    const textarea = document.createElement("textarea");
    textarea.value = text;
    textarea.setAttribute("readonly", "");
    textarea.style.position = "fixed";
    textarea.style.opacity = "0";
    document.body.appendChild(textarea);
    textarea.select();
    const ok = document.execCommand("copy");
    document.body.removeChild(textarea);
    return ok;
  } catch {
    return false;
  }
}
function ProductShareMenu({
  productId,
  productTitle,
  productPrice,
  currency,
  onCopied
}) {
  const { t } = useTranslation();
  const [open, setOpen] = useState(false);
  const [copied, setCopied] = useState(false);
  const containerRef = useRef(null);
  useEffect(() => {
    if (!open) return;
    const handlePointerDown = (e) => {
      if (containerRef.current && !containerRef.current.contains(e.target)) {
        setOpen(false);
      }
    };
    const handleKeyDown = (e) => {
      if (e.key === "Escape") {
        setOpen(false);
      }
    };
    document.addEventListener("mousedown", handlePointerDown);
    document.addEventListener("keydown", handleKeyDown);
    return () => {
      document.removeEventListener("mousedown", handlePointerDown);
      document.removeEventListener("keydown", handleKeyDown);
    };
  }, [open]);
  const shareUrl = buildProductShareUrl(productId);
  const whatsappShareText = t("share.whatsappText", {
    title: productTitle,
    price: productPrice.toLocaleString("en-US"),
    currency,
    url: shareUrl
  });
  const whatsappShareHref = `https://wa.me/?text=${encodeURIComponent(whatsappShareText)}`;
  const handleCopyLink = async () => {
    const ok = await copyTextToClipboard(shareUrl);
    if (ok) {
      setCopied(true);
      onCopied?.(productTitle);
      setTimeout(() => {
        setCopied(false);
        setOpen(false);
      }, 1200);
    }
  };
  return <div ref={containerRef} className="relative inline-block">
      <button
    type="button"
    onClick={() => setOpen((prev) => !prev)}
    aria-haspopup="menu"
    aria-expanded={open}
    aria-label={t("share.ariaLabel", { title: productTitle })}
    className={`inline-flex items-center gap-1.5 px-2.5 py-1 text-xs font-medium rounded-lg border transition-colors whitespace-nowrap shrink-0 ${copied ? "bg-pink-600/30 border-pink-400/50 text-pink-100" : "bg-white/10 hover:bg-white/20 border-pink-300/20 text-pink-100 hover:text-white"}`}
  >
        {copied ? <Check className="w-3.5 h-3.5 text-pink-300" aria-hidden="true" /> : <Share2 className="w-3.5 h-3.5" aria-hidden="true" />}
        <span>{copied ? t("share.copiedLink") : t("share.btnLabel")}</span>
      </button>

      {open && <div
    role="menu"
    className="absolute end-0 top-full mt-1.5 w-52 bg-[#1a0830] border border-pink-300/25 rounded-xl shadow-2xl shadow-purple-950/75 p-1.5 z-30 space-y-1"
  >
          <button
    type="button"
    role="menuitem"
    onClick={() => void handleCopyLink()}
    className="w-full flex items-center gap-2.5 px-3 py-2 text-xs font-medium text-pink-50 hover:bg-white/10 rounded-lg transition-colors text-start"
  >
            {copied ? <Check className="w-4 h-4 text-pink-300 shrink-0" aria-hidden="true" /> : <Copy className="w-4 h-4 text-pink-300 shrink-0" aria-hidden="true" />}
            <span>{copied ? t("share.copiedLink") : t("share.copyLink")}</span>
          </button>

          <a
    href={whatsappShareHref}
    target="_blank"
    rel="noopener noreferrer"
    role="menuitem"
    onClick={() => setOpen(false)}
    className="w-full flex items-center gap-2.5 px-3 py-2 text-xs font-medium text-pink-50 hover:bg-white/10 rounded-lg transition-colors text-start"
  >
            <MessageCircle className="w-4 h-4 text-pink-300 shrink-0" aria-hidden="true" />
            <span>{t("share.shareWhatsapp")}</span>
          </a>
        </div>}
    </div>;
}

    >
                <Star className={`${iconClass} text-amber-400 fill-amber-400`} />
              </span>}
          </span>;
  })}
    </span>;
}
function ProductRatingSummary({
  average,
  count,
  size = "sm",
  onClick
}) {
  const { t, formatReviewCount } = useTranslation();
  const textSize = size === "md" ? "text-sm" : "text-xs";
  const countText = formatReviewCount(count);
  const content = count > 0 ? <span className={`inline-flex items-center gap-2 flex-wrap ${textSize}`}>
        <StarRow rating={average} size={size} />
        <span className="font-semibold text-white tabular-nums">
          {average.toFixed(1)}
        </span>
        <span className="text-pink-200/40" aria-hidden="true">
          ·
        </span>
        <span className="text-pink-200/75 tabular-nums">
          ({countText})
        </span>
      </span> : <span className={`inline-flex items-center gap-2 ${textSize} text-pink-200/70`}>
        <StarRow rating={0} size={size} />
        <span>{t("reviews.noRatingsYet")}</span>
      </span>;
  const ariaSummary = count > 0 ? `${average.toFixed(1)} / 5 (${countText})` : t("reviews.noRatingsYet");
  if (onClick) {
    return <button
      type="button"
      onClick={onClick}
      aria-label={ariaSummary}
      className="inline-flex items-center text-start hover:opacity-80 transition-opacity focus-visible:outline-2 focus-visible:outline-pink-400 rounded"
    >
        {content}
      </button>;
  }
  return <div className="flex items-center" aria-label={ariaSummary}>
      {content}
    </div>;
}

  ProductRatingSummary,
  StarRow

import { PRESET_LANGUAGES } from "../i18n/messages";
import { useTranslation } from "../i18n/TranslationContext";
function LanguageSwitcher() {
  const {
    language,
    isTranslating,
    translateError,
    setTargetLanguage,
    clearTranslateError,
    t
  } = useTranslation();
  const [open, setOpen] = useState(false);
  const [customLanguage, setCustomLanguage] = useState("");
  const containerRef = useRef(null);
  useEffect(() => {
    if (!open) return;
    const handleOutside = (e) => {
      if (containerRef.current && !containerRef.current.contains(e.target)) {
        setOpen(false);
      }
    };
    const handleKey = (e) => {
      if (e.key === "Escape") {
        setOpen(false);
      }
    };
    document.addEventListener("mousedown", handleOutside);
    document.addEventListener("keydown", handleKey);
    return () => {
      document.removeEventListener("mousedown", handleOutside);
      document.removeEventListener("keydown", handleKey);
    };
  }, [open]);
  const handleSelectPreset = async (code, label) => {
    await setTargetLanguage(code, label);
    setOpen(false);
  };
  const handleCustomSubmit = async (e) => {
    e.preventDefault();
    const cleaned = customLanguage.trim();
    if (!cleaned) return;
    await setTargetLanguage(cleaned, cleaned);
    setCustomLanguage("");
    setOpen(false);
  };
  return <div ref={containerRef} className="relative shrink-0">
      <button
    type="button"
    onClick={() => {
      clearTranslateError();
      setOpen((prev) => !prev);
    }}
    aria-expanded={open}
    aria-haspopup="dialog"
    aria-label={t("lang.switchLabel")}
    className="inline-flex items-center gap-1.5 px-3 py-2 text-xs sm:text-sm font-medium text-pink-50 bg-white/10 border border-pink-300/25 rounded-lg hover:bg-white/20 hover:text-white transition-colors whitespace-nowrap shrink-0 min-h-[40px]"

  CheckCircle2,
  Copy,
  MessageCircle,
  Minus,
  Plus,
  ShoppingBag,
  Trash2,
  X
} from "lucide-react";
import { useTranslation } from "../i18n/TranslationContext";
import { submitOrder } from "../services/storeService";
import {
  VALIDATION_LIMITS
} from "../types/store";
import { ResilientImage } from "./ResilientImage";
function CartDrawer({
  open,
  items,
  settings,
  onClose,
  onUpdateQuantity,
  onRemoveItem,
  onClearCart,
  onOrderCreated
}) {
  const { t, td } = useTranslation();
  const [step, setStep] = useState("cart");
  const [customerName, setCustomerName] = useState("");
  const [customerPhone, setCustomerPhone] = useState("");
  const [customerCity, setCustomerCity] = useState("");
  const [customerAddress, setCustomerAddress] = useState("");
  const [notes, setNotes] = useState("");
  const [paymentMethod, setPaymentMethod] = useState("cod");
  const [submitting, setSubmitting] = useState(false);
  const [formError, setFormError] = useState("");
  const [confirmedOrderNumber, setConfirmedOrderNumber] = useState("");
  const [confirmedSummaryText, setConfirmedSummaryText] = useState("");
  const [copiedCode, setCopiedCode] = useState(false);
  if (!open) return null;
  const localizedCurrency = td(settings.currency);
  const subtotal = Number(
    items.reduce((sum, item) => sum + item.price * item.quantity, 0).toFixed(2)
  );
  const qualifiesFreeShipping = settings.freeShippingThreshold > 0 && subtotal >= settings.freeShippingThreshold;
  const effectiveShipping = items.length === 0 ? 0 : qualifiesFreeShipping ? 0 : settings.shippingFee;
  const total = Number((subtotal + effectiveShipping).toFixed(2));
  const remainingForFree = Math.max(0, settings.freeShippingThreshold - subtotal);
  const handleClose = () => {
    if (step === "confirmed") {
      setStep("cart");
    }
    setFormError("");
    onClose();
  };
  const handleSubmitOrder = async (e) => {
    e.preventDefault();
    setFormError("");
    if (customerName.trim().length < VALIDATION_LIMITS.ORDER_NAME_MIN) {
      setFormError(t("cart.errName"));
      return;
    }
    if (customerPhone.trim().length < VALIDATION_LIMITS.ORDER_PHONE_MIN) {
      setFormError(t("cart.errPhone"));
      return;
    }
    if (customerCity.trim().length < VALIDATION_LIMITS.ORDER_CITY_MIN) {
      setFormError(t("cart.errCity"));
      return;
    }
    if (customerAddress.trim().length < VALIDATION_LIMITS.ORDER_ADDRESS_MIN) {
      setFormError(t("cart.errAddress"));
      return;
    }
    if (items.length === 0) {
      setFormError(t("cart.errEmpty"));
      return;
    }
    setSubmitting(true);
    try {
      const { orderNumber } = await submitOrder({
        customerName,
        customerPhone,
        customerCity,
        customerAddress,
        notes,
        paymentMethod,
        cartItems: items,
        shippingFee: effectiveShipping
      });
      const itemsSummary = items.map((i) => `• ${i.title} (${i.variant}) × ${i.quantity}`).join("\n");
      const waMsg = `مرحباً، أرسلت طلباً جديداً رقم ${orderNumber}
الاسم: ${customerName.trim()}
المدينة: ${customerCity.trim()}
المنتجات:
${itemsSummary}
الإجمالي: ${total} ${settings.currency}`;
      setConfirmedOrderNumber(orderNumber);
      setConfirmedSummaryText(waMsg);
      onOrderCreated(orderNumber);
      onClearCart();
      setCustomerName("");
      setCustomerPhone("");
      setCustomerCity("");
      setCustomerAddress("");
      setNotes("");
      setStep("confirmed");
    } catch {
      setFormError(t("cart.errSubmit"));
    } finally {
      setSubmitting(false);
    }
  };
  const cleanWhatsapp = settings.whatsapp.replace(/[^0-9]/g, "");
  const whatsappOrderHref = cleanWhatsapp ? `https://wa.me/${cleanWhatsapp}?text=${encodeURIComponent(confirmedSummaryText)}` : void 0;
  const handleCopyCode = async () => {
    try {
      await navigator.clipboard.writeText(confirmedOrderNumber);
      setCopiedCode(true);
      setTimeout(() => setCopiedCode(false), 2e3);
    } catch {
    }
  };
  return <div
    className="fixed inset-0 z-50 flex justify-end bg-[#0d0318]/80 backdrop-blur-sm"
    role="dialog"
    aria-modal="true"
    aria-label={t("cart.dialogLabel")}
  >
      <div className="w-full max-w-md bg-[#18082b] text-white h-full flex flex-col border-s border-pink-300/25 shadow-2xl">
        {
    /* Drawer Header */
  }
        <div className="px-6 py-4 border-b border-pink-300/15 flex items-center justify-between gap-4">
          <div className="flex items-center gap-2">
            <ShoppingBag className="w-5 h-5 text-pink-300" aria-hidden="true" />
            <h2 className="text-lg font-semibold text-white">
              {step === "cart" && t("cart.titleCart")}
              {step === "checkout" && t("cart.titleCheckout")}
              {step === "confirmed" && t("cart.titleConfirmed")}
            </h2>
          </div>
          <button
    type="button"
    onClick={handleClose}
    aria-label={t("cart.closeLabel")}
    className="w-10 h-10 inline-flex items-center justify-center rounded-lg text-pink-100 hover:text-white hover:bg-white/10 transition-colors"
  >
            <X className="w-5 h-5" aria-hidden="true" />
          </button>
        </div>

        {
    /* Drawer Body */
  }
        <div className="flex-1 overflow-y-auto p-6">
          {step === "confirmed" ? <div className="py-6 space-y-6 text-center">
              <div className="w-14 h-14 mx-auto rounded-full bg-pink-500/20 text-pink-300 flex items-center justify-center">
                <CheckCircle2 className="w-8 h-8" aria-hidden="true" />
              </div>
              <div className="space-y-2">
                <h3 className="font-serif text-2xl font-bold text-white">
                  {t("cart.confirmedHeading")}
                </h3>
                <p className="text-sm text-pink-100/80 leading-relaxed">
                  {t("cart.confirmedDesc")}
                </p>
              </div>

              <div className="p-4 bg-[#120520]/80 border border-pink-300/20 rounded-xl space-y-2">
                <span className="block text-xs text-pink-200/75">
                  {t("cart.trackingCodeLabel")}
                </span>
                <div className="flex items-center justify-center gap-2">
                  <span translate="no" className="text-lg font-semibold text-white tabular-nums tracking-wide">
                    {confirmedOrderNumber}
                  </span>
                  <button
    type="button"
    onClick={handleCopyCode}
    aria-label={t("cart.copyCodeLabel")}
    className="inline-flex items-center gap-1 px-2.5 py-1 text-xs font-medium text-pink-100 bg-white/10 border border-pink-300/20 rounded-md hover:bg-white/20"
  >
                    <Copy className="w-3.5 h-3.5" aria-hidden="true" />
                    <span>{copiedCode ? t("cart.copied") : t("cart.copy")}</span>
                  </button>
                </div>
              </div>

              <div className="space-y-3 pt-2">
                {whatsappOrderHref && <a
    href={whatsappOrderHref}
    target="_blank"
    rel="noopener noreferrer"
    className="w-full inline-flex items-center justify-center gap-2 px-5 py-3 text-sm font-semibold text-white bg-gradient-to-l from-pink-600 to-fuchsia-600 rounded-lg hover:from-pink-500 hover:to-fuchsia-500 transition-colors min-h-[44px]"
  >
                    <MessageCircle className="w-4 h-4" aria-hidden="true" />
                    <span>{t("cart.sendWhatsappOrder")}</span>
                  </a>}
                <button
    type="button"
    onClick={handleClose}
    className="w-full px-5 py-2.5 text-sm font-medium text-pink-50 bg-white/10 rounded-lg hover:bg-white/20 transition-colors min-h-[42px]"
  >
                  {t("cart.continueShopping")}
                </button>
              </div>
            </div> : items.length === 0 ? <div className="h-full flex flex-col items-center justify-center text-center py-12 space-y-4">
              <div className="w-14 h-14 rounded-full bg-white/10 text-pink-300/70 flex items-center justify-center">
                <ShoppingBag className="w-6 h-6" aria-hidden="true" />
              </div>
              <div className="space-y-1">
                <p className="text-base font-medium text-white">
                  {t("cart.emptyTitle")}

                  {t("cart.emptySubtitle")}

import { Component } from "react";
class ErrorBoundary extends Component {
  state = {
    hasError: false,
    message: ""
  };
  static getDerivedStateFromError(error) {
    let readable = error.message;
    try {
      const parsed = JSON.parse(error.message);
      if (parsed && parsed.error) {
        readable = parsed.error;
      }
    } catch {
    }
    return { hasError: true, message: readable };
  }
  componentDidCatch(error, errorInfo) {
    console.error("Application boundary caught error:", error, errorInfo);
  }
  render() {
    if (this.state.hasError) {
      return <div
        dir="rtl"
        className="min-h-screen text-white flex items-center justify-center p-6"
      >
          <div className="max-w-md w-full bg-[#1b0a31]/90 backdrop-blur-md border border-pink-300/20 rounded-xl p-8 text-center space-y-4">
            <h2 className="text-xl font-semibold text-white">
              حدث خلل غير متوقع في تحميل البيانات
            </h2>
            <p className="text-sm text-pink-100/80 leading-relaxed">
              {this.state.message || "يرجى تحديث الصفحة والمحاولة مرة أخرى."}
            </p>
            <button
        type="button"
        onClick={() => window.location.reload()}
        className="px-5 py-2.5 text-sm font-medium text-white bg-gradient-to-l from-pink-600 to-fuchsia-600 rounded-lg hover:from-pink-500 hover:to-fuchsia-500 transition-colors"
      >
              إعادة تحميل الصفحة
            </button>
          </div>
        </div>;
    }
    return this.props.children;
  }
}
export {
  ErrorBoundary
};

    fallbackTitle={settings.storeName}
    className="w-full h-full object-contain"
  />
                    </div>
                  </div>
                </div>
              </div>

              {
    /* 4-column Store Guarantees Strip below hero split */
  }
              <div className="border-t border-pink-300/15 bg-[#140624]/55 backdrop-blur-sm">
                <div className="max-w-6xl mx-auto px-4 sm:px-6 lg:px-8 py-5 grid grid-cols-2 lg:grid-cols-4 gap-6 text-sm">
                  <div className="space-y-0.5">
                    <p className="font-semibold text-white">
                      {t("hero.guarantee1Title")}
                    </p>
                    <p className="text-xs text-pink-200/75">
                      {t("hero.guarantee1Desc")}
                    </p>
                  </div>
                  <div className="space-y-0.5">
                    <p className="font-semibold text-white">
                      {t("hero.guarantee2Title")}
                    </p>
                    <p className="text-xs text-pink-200/75">
                      {t("hero.guarantee2Desc")}
                    </p>
                  </div>
                  <div className="space-y-0.5">
                    <p className="font-semibold text-white">
                      {t("hero.guarantee3Title")}
                    </p>
                    <p className="text-xs text-pink-200/75 tabular-nums">
                      {t("hero.guarantee3Desc", {
    amount: settings.freeShippingThreshold.toLocaleString("en-US"),
    currency: localizedCurrency
  })}
                    </p>
                  </div>
                  <div className="space-y-0.5">
                    <p className="font-semibold text-white">
                      {t("hero.guarantee4Title")}
                    </p>
                    <p className="text-xs text-pink-200/75">
                      {t("hero.guarantee4Desc")}
                    </p>
                  </div>
                </div>
              </div>
            </section>

            {
    /* Section 2: Featured Collection & Catalog Grid */
  }
            <section
    id="catalog"
    className="max-w-6xl mx-auto px-4 sm:px-6 lg:px-8 py-14 sm:py-20 space-y-8"
  >
              <div className="flex flex-col md:flex-row md:items-end justify-between gap-6">
                <div className="space-y-2">
                  <h2 className="font-serif text-3xl font-bold text-white">
                    {t("catalog.heading")}
                  </h2>
                  <p className="text-sm text-pink-100/80">
                    {t("catalog.subheading")}
                  </p>
                </div>

                {
    /* Search Input */
  }
                <div className="relative w-full md:w-72">
                  <Search
    className="w-4 h-4 text-pink-200/60 absolute start-3.5 top-1/2 -translate-y-1/2 pointer-events-none"
    aria-hidden="true"
  />
                  <input
    type="search"
    value={searchQuery}
    onChange={(e) => setSearchQuery(e.target.value)}
    placeholder={t("catalog.searchPlaceholder")}
    aria-label={t("catalog.searchLabel")}
    className="w-full ps-10 pe-4 py-2.5 text-sm bg-[#120520]/80 border border-pink-300/25 text-white placeholder:text-pink-200/45 rounded-lg focus:outline-none focus:border-pink-400"
  />
                </div>
              </div>

              {
    /* Interactive Category Filter Tabs */
  }
              <div
    role="tablist"
    aria-label={t("catalog.categoriesLabel")}
    className="flex flex-wrap items-center gap-2 p-1.5 bg-[#140624]/70 border border-pink-300/20 backdrop-blur-sm rounded-xl w-fit"
  >
                {categories.map((cat) => {
    const active = selectedCategory === cat;
    const label = cat === ALL_CATEGORY_KEY ? t("catalog.allCategory") : td(cat);
    return <button
      key={cat}
      type="button"
      role="tab"
      aria-selected={active}
      onClick={() => setSelectedCategory(cat)}
      className={`px-4 py-2 text-xs sm:text-sm font-medium rounded-lg transition-colors whitespace-nowrap shrink-0 ${active ? "bg-pink-600 text-white shadow-sm" : "text-pink-100/80 hover:text-white"}`}
    >
                      {label}
                    </button>;
  })}
              </div>

              {
    /* 3-Column Product Grid */
  }
              {filteredProducts.length === 0 ? <div className="bg-[#1b0a31]/85 backdrop-blur-md border border-pink-300/20 rounded-xl p-12 text-center space-y-3">
                  <p className="text-base font-medium text-white">
                    {t("catalog.emptySearch")}
                  </p>
                  <button
    type="button"
    onClick={() => {
      setSelectedCategory(ALL_CATEGORY_KEY);
      setSearchQuery("");
    }}
    className="px-4 py-2 text-xs font-medium text-pink-200 bg-white/10 hover:bg-white/20 border border-pink-300/20 rounded-lg transition-colors"
  >
                    {t("catalog.showAll")}
                  </button>
                </div> : <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-8">
                  {filteredProducts.map((product) => {
    const defaultVariant = product.variants.length > 0 ? product.variants[0] : "قياسي";
    const outOfStock = product.stock <= 0;
    const ratingStats = ratingStatsByProduct[product.id] ?? {
      average: 0,
      count: 0
    };
    const localizedTitle = td(product.title);
    const localizedCategory = td(product.category);
    const localizedDescription = td(product.description);
    return <article
      key={product.id}
      className="relative group bg-[#1b0a31]/85 backdrop-blur-md border border-pink-300/20 rounded-xl flex flex-col transition-all duration-200 hover:-translate-y-0.5 hover:z-20 focus-within:z-20 hover:border-pink-400/50 hover:shadow-xl hover:shadow-purple-950/50"
    >
                        <button
      type="button"
      onClick={() => setSelectedProduct(product)}
      className="block w-full text-start aspect-[4/3] bg-[#120520] rounded-t-xl overflow-hidden focus-visible:outline-2 focus-visible:outline-pink-400"
    >
                          <ResilientImage
      src={product.imageUrl}
      alt={localizedTitle}
      fallbackTitle={localizedTitle}
      className="w-full h-full object-cover transition-transform duration-300 group-hover:scale-[1.03]"
    />
                        </button>

                        <div className="p-5 flex-1 flex flex-col justify-between gap-4">
                          <div className="space-y-1.5">
                            <div className="flex items-center justify-between gap-2 text-xs text-pink-200/75">
                              <div className="flex items-center gap-2 min-w-0">
                                <span className="truncate">{localizedCategory}</span>
                                <span aria-hidden="true">·</span>
                                <span className="whitespace-nowrap">
                                  {outOfStock ? t("catalog.outOfStockShort") : t("catalog.inStockShort")}
                                </span>
                              </div>
                              <ProductShareMenu
      productId={product.id}
      productTitle={localizedTitle}
      productPrice={product.price}
      currency={localizedCurrency}
    />
                            </div>

                            <button
      type="button"
      onClick={() => setSelectedProduct(product)}
      className="block text-start w-full"
    >
                              <h3 className="text-base font-semibold text-white group-hover:text-pink-300 transition-colors leading-snug">
                                {localizedTitle}
                              </h3>
                            </button>

                            <ProductRatingSummary
      average={ratingStats.average}
      count={ratingStats.count}
      size="sm"
    />

                            <p className="text-sm text-pink-100/80 line-clamp-2 leading-relaxed">
                              {localizedDescription}
                            </p>
                          </div>

                          <div className="flex items-center justify-between gap-3 pt-3 border-t border-pink-300/15">
                            <span className="text-base font-bold text-pink-300 tabular-nums">
                              {product.price.toLocaleString("en-US")} {localizedCurrency}
                            </span>

                            <div className="flex items-center gap-2">
                              <button
      type="button"
      onClick={() => setSelectedProduct(product)}
      className="px-3 py-2 text-xs font-medium text-pink-100 hover:text-white bg-white/10 hover:bg-white/20 border border-pink-300/15 rounded-lg transition-colors whitespace-nowrap shrink-0"
    >
                                {t("catalog.detailsBtn")}
                              </button>
                              <button
      type="button"
      disabled={outOfStock}
      onClick={() => handleAddToCart(product, defaultVariant, 1)}
      className="inline-flex items-center gap-1.5 px-3.5 py-2 text-xs font-semibold text-white bg-gradient-to-l from-pink-600 to-fuchsia-600 rounded-lg hover:from-pink-500 hover:to-fuchsia-500 disabled:bg-white/10 disabled:from-transparent disabled:to-transparent disabled:text-pink-200/40 transition-all whitespace-nowrap shrink-0"
    >
                                <ShoppingBag className="w-3.5 h-3.5" aria-hidden="true" />
                                <span>
                                  {outOfStock ? t("catalog.soldOutBtn") : t("catalog.addToCart")}
                                </span>
                              </button>
                            </div>
                          </div>
                        </div>
                      </article>;
  })}
                </div>}
            </section>

            {
    /* Section 3: Story & Direct Customer Contact */
  }
            <section
    id="contact"
    className="border-t border-pink-300/15 bg-[#120520]/30 py-14 sm:py-20"
  >
              <div className="max-w-6xl mx-auto px-4 sm:px-6 lg:px-8">
                <div className="grid grid-cols-1 lg:grid-cols-12 gap-10 lg:gap-12 items-start">
                  {
    /* Story & Direct Channels */
  }
                  <div id="story" className="lg:col-span-5 space-y-6">
                    <div className="space-y-3">
                      <h2 className="font-serif text-3xl font-bold text-white">
                        {t("contact.heading")}
                      </h2>
                      <p className="text-sm sm:text-base text-pink-100/85 leading-relaxed">
                        {t("contact.storyBody", { storeName: settings.storeName })}
                      </p>
                    </div>

                    <div className="space-y-3 pt-2 border-t border-pink-300/15 text-sm">
                      {directWhatsappHref && <a
    href={directWhatsappHref}
    target="_blank"
    rel="noopener noreferrer"
    className="flex items-center justify-between p-3.5 rounded-xl bg-gradient-to-l from-pink-600 to-fuchsia-600 text-white hover:from-pink-500 hover:to-fuchsia-500 shadow-lg shadow-pink-950/30 transition-all"
  >
                          <span className="inline-flex items-center gap-2.5 font-semibold">
                            <MessageCircle className="w-4 h-4" aria-hidden="true" />
                            <span>{t("contact.whatsappDirect")}</span>
                          </span>
                          <span translate="no" className="text-xs opacity-90 tabular-nums" dir="ltr">
                            +{cleanWhatsapp}
                          </span>
                        </a>}

                      {settings.phone && <a
    href={`tel:${settings.phone}`}
    className="flex items-center gap-3 p-3 rounded-xl bg-[#1b0a31]/85 border border-pink-300/20 text-pink-50 hover:bg-[#260f42] transition-colors"
  >
                          <Phone className="w-4 h-4 text-pink-400 shrink-0" aria-hidden="true" />
                          <span translate="no" className="tabular-nums" dir="ltr">
                            {settings.phone}
                          </span>
                        </a>}

                      {settings.email && <a
    href={`mailto:${settings.email}`}
    className="flex items-center gap-3 p-3 rounded-xl bg-[#1b0a31]/85 border border-pink-300/20 text-pink-50 hover:bg-[#260f42] transition-colors"
  >
                          <Mail className="w-4 h-4 text-pink-400 shrink-0" aria-hidden="true" />
                          <span translate="no" className="truncate">{settings.email}</span>
                        </a>}

                      {instagramHref && <a
    href={instagramHref}
    target="_blank"
    rel="noopener noreferrer"
    className="flex items-center justify-between p-3 rounded-xl bg-[#1b0a31]/85 border border-pink-300/20 text-pink-50 hover:bg-[#260f42] transition-colors"
  >
                          <span className="inline-flex items-center gap-3 font-medium">
                            <Instagram className="w-4 h-4 text-pink-400 shrink-0" aria-hidden="true" />
                            <span>{t("contact.instagram")}</span>
                          </span>
                          <span translate="no" className="text-xs text-pink-200/80 font-medium" dir="ltr">
                            @{instagramHandle}
                          </span>
                        </a>}

                      <div className="flex items-center gap-3 px-3 py-1.5 text-xs text-pink-200/75">
                        <MapPin className="w-4 h-4 text-pink-300/70 shrink-0" aria-hidden="true" />
                        <span>{localizedCity}</span>
                      </div>

                      <div className="flex items-center gap-3 px-3 py-1.5 text-xs text-pink-200/75">
                        <Clock className="w-4 h-4 text-pink-300/70 shrink-0" aria-hidden="true" />
                        <span>{localizedHours}</span>
                      </div>
                    </div>
                  </div>

                  {
    /* Customer Message Form */
  }
                  <div className="lg:col-span-7 bg-[#1b0a31]/85 backdrop-blur-md border border-pink-300/20 rounded-xl p-6 sm:p-8">
                    <div className="space-y-1 mb-6">
                      <h3 className="text-xl font-semibold text-white">
                        {t("contact.formTitle")}
                      </h3>
                      <p className="text-xs sm:text-sm text-pink-200/75">
                        {t("contact.formSubtitle")}
                      </p>
                    </div>

                    {messageSent ? <div className="p-6 bg-[#140624]/85 border border-pink-300/20 rounded-xl text-center space-y-3">
                        <CheckCircle2
    className="w-10 h-10 text-pink-400 mx-auto"
    aria-hidden="true"
  />
                        <h4 className="text-base font-semibold text-white">
                          {t("contact.sentTitle")}
                        </h4>
                        <p className="text-sm text-pink-100/80">
                          {t("contact.sentBody", { storeName: settings.storeName })}
                        </p>
                        <button
    type="button"
    onClick={() => setMessageSent(false)}
    className="px-4 py-2 text-xs font-medium text-pink-100 bg-white/10 hover:bg-white/20 border border-pink-300/20 rounded-lg transition-colors"
  >
                          {t("contact.sendAnother")}
                        </button>
                      </div> : <form onSubmit={handleSendContactMessage} className="space-y-4">
                        {messageError && <div className="p-3 bg-red-950/60 border border-red-400/40 rounded-lg text-xs text-red-200">
                            {messageError}
                          </div>}

                        <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                          <div>
                            <label className="block text-xs font-medium text-pink-100/90 mb-1.5">
                              {t("contact.nameLabel")}
                            </label>
                            <input
    type="text"
    required
    maxLength={VALIDATION_LIMITS.MSG_NAME_MAX}
    value={senderName}
    onChange={(e) => setSenderName(e.target.value)}
    placeholder={t("contact.namePlaceholder")}
    className="w-full px-3.5 py-2.5 text-sm bg-[#120520]/90 border border-pink-300/25 text-white placeholder:text-pink-200/45 rounded-lg focus:outline-none focus:border-pink-400"
  />
                          </div>

                          <div>
                            <label className="block text-xs font-medium text-pink-100/90 mb-1.5">
                              {t("contact.contactLabel")}
                            </label>
                            <input
    type="text"
    required
    maxLength={VALIDATION_LIMITS.MSG_CONTACT_MAX}
    value={senderContact}
    onChange={(e) => setSenderContact(e.target.value)}
    placeholder={t("contact.contactPlaceholder")}
    className="w-full px-3.5 py-2.5 text-sm bg-[#120520]/90 border border-pink-300/25 text-white placeholder:text-pink-200/45 rounded-lg focus:outline-none focus:border-pink-400"
  />
                          </div>
                        </div>

                        <div>
                          <label className="block text-xs font-medium text-pink-100/90 mb-1.5">
                            {t("contact.subjectLabel")}
                          </label>
                          <input
    type="text"
    required
    maxLength={VALIDATION_LIMITS.MSG_SUBJECT_MAX}
    value={subject}
    onChange={(e) => setSubject(e.target.value)}
    placeholder={t("contact.subjectPlaceholder")}
    className="w-full px-3.5 py-2.5 text-sm bg-[#120520]/90 border border-pink-300/25 text-white placeholder:text-pink-200/45 rounded-lg focus:outline-none focus:border-pink-400"
  />
                        </div>

                        <div>
                          <label className="block text-xs font-medium text-pink-100/90 mb-1.5">
                            {t("contact.bodyLabel")}
                          </label>
                          <textarea
    required
    rows={4}
    maxLength={VALIDATION_LIMITS.MSG_BODY_MAX}
    value={body}
    onChange={(e) => setBody(e.target.value)}
    placeholder={t("contact.bodyPlaceholder")}
    className="w-full px-3.5 py-2.5 text-sm bg-[#120520]/90 border border-pink-300/25 text-white placeholder:text-pink-200/45 rounded-lg focus:outline-none focus:border-pink-400"
  />
                        </div>

                        <button
    type="submit"
    disabled={sendingMessage}
    className="w-full sm:w-auto px-7 py-3 text-sm font-semibold text-white bg-gradient-to-l from-pink-600 to-fuchsia-600 rounded-lg hover:from-pink-500 hover:to-fuchsia-500 disabled:opacity-60 transition-all min-h-[44px]"
  >
                          {sendingMessage ? t("contact.sending") : t("contact.sendBtn")}
                        </button>
                      </form>}
                  </div>
                </div>
              </div>
            </section>
          </>}
      </main>

      {
    /* Quiet Footer */
  }
      <footer className="border-t border-pink-300/15 bg-[#140624]/75 backdrop-blur-sm py-8">
        <div className="max-w-6xl mx-auto px-4 sm:px-6 lg:px-8 flex flex-col sm:flex-row items-center justify-between gap-4 text-xs text-pink-200/75">
          <p>
            {t("footer.rights", {
    year: (/* @__PURE__ */ new Date()).getFullYear(),
    storeName: settings.storeName
  })}
          </p>
          <div className="flex flex-wrap items-center gap-6">
            <button
    type="button"
    onClick={() => handleNavigateSection("catalog")}
    className="hover:text-white transition-colors"
  >
              {t("nav.products")}
            </button>
            <button
    type="button"
    onClick={() => setTrackerOpen(true)}
    className="hover:text-white transition-colors"
  >
              {t("nav.trackOrder")}
            </button>
            <button
    type="button"
    onClick={() => handleNavigateSection("contact")}
    className="hover:text-white transition-colors"
  >
              {t("nav.contact")}
            </button>
            {instagramHref && <a
    href={instagramHref}
    target="_blank"
    rel="noopener noreferrer"
    className="inline-flex items-center gap-1.5 hover:text-white transition-colors"
  >
                <Instagram className="w-3.5 h-3.5" aria-hidden="true" />
                <span translate="no" dir="ltr">@{instagramHandle}</span>
              </a>}
            <button
    type="button"
    onClick={() => {
      setView("admin");
      window.scrollTo({ top: 0, behavior: "smooth" });
    }}
    className="text-pink-300 font-medium hover:text-pink-200 hover:underline"
  >
              {t("nav.storeAdmin")}
            </button>
          </div>
        </div>
      </footer>

      {
    /* Modals & Drawers */
  }
      {selectedProduct && <ProductDetailModal
    product={selectedProduct}
    currency={settings.currency}
    whatsapp={settings.whatsapp}
    reviews={reviewsByProduct[selectedProduct.id] ?? []}
    ratingStats={ratingStatsByProduct[selectedProduct.id] ?? { average: 0, count: 0 }}
    onClose={handleCloseProductModal}
    onAddToCart={handleAddToCart}
  />}

      <CartDrawer
    open={cartOpen}
    items={cart}
    settings={settings}
    onClose={() => setCartOpen(false)}
    onUpdateQuantity={handleUpdateCartQuantity}
    onRemoveItem={handleRemoveCartItem}
    onClearCart={() => setCart([])}
    onOrderCreated={handleOrderCreated}
  />

      <OrderTrackerModal
    open={trackerOpen}
    currency={settings.currency}
    recentOrders={recentOrders}
    onClose={() => setTrackerOpen(false)}
  />

      <ScrollToTopButton
    raised={Boolean(cartNotice)}
    hidden={Boolean(selectedProduct || cartOpen || trackerOpen)}
  />
    </div>;
}
function App() {
  return <ErrorBoundary>
      <TranslationProvider>
        <StorefrontApp />
      </TranslationProvider>
    </ErrorBoundary>;
}
export {
  App as default
};
